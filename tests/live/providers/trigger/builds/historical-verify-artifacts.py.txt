#!/usr/bin/env python3
"""Verify source-built S05 runtimes without starting a server or editing jars."""
from pathlib import Path
import hashlib, json, subprocess, zipfile

ROOT=Path(__file__).resolve().parent
PATCHED='core/src/main/java/net/momirealms/craftengine/core/item/updater/impl/ResetOperation.java'
RESET='net/momirealms/craftengine/core/item/updater/impl/ResetOperation.class'
sha=lambda b:hashlib.sha256(b).hexdigest()
checks=[]
for lane,version,commit in (
    ('ce-26.9.1-2b9e6f59','26.9.1','2b9e6f59a0bf4dac7f71958c0ea96c14b8e44985'),
    ('sf-26.9.2-94b74598','26.9.2','94b74598312d5a16216caed1c07fe3966d854096'),
):
    d=ROOT/lane
    s=d/'source'
    git=lambda *args:subprocess.check_output(['git','-C',str(s),*args],text=True).strip()
    assert git('rev-parse','HEAD')==commit
    assert git('diff','--name-only').splitlines()==[PATCHED]
    assert git('diff','--numstat')==f'1\t1\t{PATCHED}'
    assert not git('ls-files','--others','--exclude-standard')
    diff=subprocess.check_output(['git','-C',str(s),'diff','--binary'])
    (d/'source-changes.patch').write_bytes(diff)
    jar=s/f'target/craft-engine-paper-plugin-{version}.jar'
    info=json.loads((d/'provenance.json').read_text())
    assert sha(jar.read_bytes())==info['artifact']['sha256']
    with zipfile.ZipFile(jar) as z:
        assert z.testzip() is None
        names=z.namelist()
        assert 'plugin.yml' not in names
        descriptor=z.read('paper-plugin.yml')
        assert b'name: CraftEngine' in descriptor
        assert f'version: {version}'.encode() in descriptor
        assert b'main: net.momirealms.craftengine.bukkit.plugin.PaperCraftEnginePlugin' in descriptor
        assert b'bootstrapper: net.momirealms.craftengine.bukkit.plugin.PaperCraftEngineBootstrap' in descriptor
        generated=list((s/'bukkit/paper-loader/build').rglob('paper-plugin.yml'))
        assert any(p.read_bytes()==descriptor for p in generated)
        assert b'paperweight-mappings-namespace: mojang' in z.read('META-INF/MANIFEST.MF')
        assert f'git-version={commit[:8]}'.encode() in z.read('craft-engine.properties')
        assert z.read(RESET)==(s/'core/build/classes/java/main'/RESET).read_bytes()
        assert int.from_bytes(z.read(RESET)[6:8],'big')==65
        assert 'net/momirealms/craftengine/bukkit/compatibility/BukkitCompatibilityManager.class' in names
        assert 'net/momirealms/craftengine/bukkit/plugin/PaperCraftEnginePlugin.class' in names
        assert 'net/momirealms/craftengine/bukkit/plugin/PaperCraftEngineBootstrap.class' in names
        proxy=z.read('proxy.jarinjar')
        info['archive_verification']={'zip_crc':'pass','native_descriptor_equals_build_generated_descriptor':True,'reset_class_equals_fresh_compiler_output':True,'reset_class_java_major_version':65,'embedded_proxy_sha256':sha(proxy),'reset_class_sha256':sha(z.read(RESET)),'classes':sum(n.endswith('.class') for n in names),'entries':len(names)}
    bytecode=(d/'ResetOperation.javap.txt').read_text()
    assert 'Item.count:()I' in bytecode
    assert 'buildItem:(Lnet/momirealms/craftengine/core/item/ItemBuildContext;I)' in bytecode
    info['archive_verification']['count_preserving_bytecode']='Item.count() -> ItemDefinition.buildItem(ItemBuildContext,int)'
    info['source_changes_patch_sha256']=sha(diff)
    info['build_changes'][0]['sha256']=sha((ROOT/'compatibility-gson-first.init.gradle').read_bytes())
    info['tracked_jar_inputs']=[{'path':p,'git_blob':git('rev-parse',f'HEAD:{p}'),'sha256':sha((s/p).read_bytes())} for p in git('ls-files','*.jar').splitlines() if p!='gradle/wrapper/gradle-wrapper.jar']
    info['gradle_version']='9.7.1'
    info['toolchain_version']=25
    info['compiler_release']=21
    info['build_metadata_changes']=['Isolated clone git user.name set to Allium S05 validation, recorded in craft-engine.properties builder field','Native build timestamp retained; git-version records exact requested upstream source commit']
    reports=sorted((d/'resolved-inputs').glob('*.json'))
    info['resolved_dependency_reports']=[{'path':str(p),'sha256':sha(p.read_bytes())} for p in reports]
    (d/'provenance.json').write_text(json.dumps(info,indent=2)+'\n')
    checks.append({'lane':lane,'jar':str(jar),'sha256':info['artifact']['sha256'],'source_commit':commit,'verification':'pass'})
(ROOT/'verification.json').write_text(json.dumps(checks,indent=2)+'\n')
(ROOT/'SHA256SUMS').write_text(''.join(f"{c['sha256']}  {Path(c['jar']).relative_to(ROOT)}\n" for c in checks))
print(json.dumps(checks,indent=2))
